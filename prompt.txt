socket.on('opportunityUpdated', (updatedOpportunity) => {

  // Update opportunities cache
  queryClient.setQueryData(['opportunities'], old =>
    old?.map(o =>
      o.id === updatedOpportunity.id ? updatedOpportunity : o
    )
  )

  // Update applications cache
  queryClient.setQueryData(['applications'], old =>
    old?.map(app =>
      app.opportunityId === updatedOpportunity.id
        ? { ...app, opportunity: updatedOpportunity }
        : app
    )
  )
})
